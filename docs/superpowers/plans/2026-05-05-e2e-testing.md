# E2E Testing Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** End-to-end tests met Playwright tegen productie Firebase, met automatische auth via Firebase Admin SDK custom tokens.

**Architecture:** Playwright draait als apart test commando (`npm run test:e2e`). Een `global-setup.js` maakt via Firebase Admin een custom auth token voor `kevin@bloxit.be`, injecteert dat in de browser, en slaat de auth state op. Alle tests hergebruiken die state. Elke test suite maakt een eigen project aan (E2E_ prefix) en ruimt het op via soft + hard delete.

**Tech Stack:** `@playwright/test`, `firebase-admin`, Chromium

---

### Task 2: Global setup — Firebase Admin auth injection

**Files:**
- Create: `e2e/global-setup.js`

- [ ] **Step 1: Create global-setup.js**

```js
// e2e/global-setup.js
import { chromium } from '@playwright/test';
import { readFileSync } from 'fs';
import { createRequire } from 'module';

const require = createRequire(import.meta.url);

export default async function globalSetup() {
  // 1. Init Firebase Admin
  const admin = require('firebase-admin');
  const serviceAccount = JSON.parse(
    readFileSync(new URL('./service-account-key.json', import.meta.url), 'utf8')
  );
  if (!admin.apps.length) {
    admin.initializeApp({
      credential: admin.credential.cert(serviceAccount),
    });
  }

  // 2. Create custom token for whitelisted email
  const customToken = await admin.auth().createCustomToken('e2e-test-user', {
    email: 'kevin@bloxit.be',
  });

  // 3. Launch browser and sign in
  const browser = await chromium.launch();
  const context = await browser.newContext();
  const page = await context.newPage();

  const baseURL = process.env.BASE_URL || 'http://localhost:8000';
  await page.goto(`${baseURL}/dashboard.html`);

  // Wait for Firebase SDK to load
  await page.waitForFunction(() => typeof firebase !== 'undefined' && firebase.auth, {
    timeout: 15_000,
  });

  // Sign in with custom token
  await page.evaluate(async (token) => {
    await firebase.auth().signInWithCustomToken(token);
  }, customToken);

  // Wait for auth state to settle
  await page.waitForFunction(
    () => firebase.auth().currentUser !== null,
    { timeout: 10_000 },
  );

  // 4. Save auth state
  await context.storageState({ path: new URL('./auth-state.json', import.meta.url).pathname });

  await browser.close();

  // Cleanup Admin SDK
  await admin.app().delete();
}
```

```js
// e2e/helpers/project-helpers.js
import { expect } from '@playwright/test';

/**
 * Create a new project via the project-edit form.
 * Returns the projectId extracted from the URL after save.
 */
export async function createTestProject(page, opts = {}) {
  const customerName = opts.customerName || `E2E_Test_${Date.now()}`;

  await page.goto('/project-edit.html?new=1');
  await page.waitForSelector('#fCustomerName', { state: 'visible', timeout: 15_000 });

  await page.fill('#fCustomerName', customerName);

  // Click "Opslaan"
  await page.click('#btnSave');

  // Wait for navigation to ?project=<id> (save redirects)
  await page.waitForURL(/project=/, { timeout: 15_000 });

  const url = new URL(page.url());
  const projectId = url.searchParams.get('project');
  if (!projectId) throw new Error('createTestProject: no projectId in URL after save');

  return { projectId, customerName };
}

/**
 * Upload a CSV file to an existing project via the project-edit form.
 */
export async function uploadCsvToProject(page, projectId, csvPath) {
  await page.goto(`/project-edit.html?project=${projectId}`);
  await page.waitForSelector('#fCsv', { state: 'attached', timeout: 15_000 });

  await page.setInputFiles('#fCsv', csvPath);

  // Wait for CSV parsing feedback
  await page.waitForFunction(
    () => {
      const el = document.getElementById('fCsvStatus');
      return el && el.textContent && !el.textContent.includes('Geen');
    },
    { timeout: 10_000 },
  );

  // Save
  await page.click('#btnSave');

  // Wait for save to complete (toast or URL stable)
  await page.waitForTimeout(2000);
}

/**
 * Cleanup a test project: soft delete → hard delete.
 * Navigates to dashboard, finds the project, and performs both deletes.
 */
export async function cleanupProject(page, projectId, customerName) {
  await page.goto('/dashboard.html');
  await page.waitForSelector('#projectSearch', { state: 'visible', timeout: 15_000 });

  // Search for the project
  await page.fill('#projectSearch', customerName);
  await page.waitForTimeout(500);

  // Soft delete
  const deleteBtn = page.locator(`.deleteBtn[data-id="${projectId}"]`);
  if (await deleteBtn.isVisible({ timeout: 3000 }).catch(() => false)) {
    // Accept the confirm dialog
    page.once('dialog', dialog => dialog.accept());
    await deleteBtn.click();
    await page.waitForTimeout(1000);
  }

  // Enable "Toon verwijderde"
  const showDeletedCheckbox = page.locator('#toggleShowDeleted');
  if (!(await showDeletedCheckbox.isChecked())) {
    await showDeletedCheckbox.check();
    await page.waitForTimeout(1000);
  }

  // Search again (list may have re-rendered)
  await page.fill('#projectSearch', customerName);
  await page.waitForTimeout(500);

  // Hard delete
  const hardDeleteBtn = page.locator(`.permdelBtn[data-id="${projectId}"]`);
  if (await hardDeleteBtn.isVisible({ timeout: 3000 }).catch(() => false)) {
    page.once('dialog', dialog => dialog.accept());
    await hardDeleteBtn.click();
    await page.waitForTimeout(1000);
  }

  // Verify project is gone
  await page.fill('#projectSearch', customerName);
  await page.waitForTimeout(500);
  const remaining = page.locator(`.projectNameBtn[data-id="${projectId}"]`);
  await expect(remaining).toHaveCount(0);
}
```
